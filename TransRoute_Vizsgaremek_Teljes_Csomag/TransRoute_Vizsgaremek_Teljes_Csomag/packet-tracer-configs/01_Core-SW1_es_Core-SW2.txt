! ============================================================
! TRANSROUTE LOGISZTIKA KFT. - CORE-SW1 & CORE-SW2 (3560-24PS)
! Felelős: Háda Patrik
! Funkciók: VLAN 10,20,30,40, Trunk, EtherChannel (LACP), Rapid-PVST
! ============================================================

! --- CORE-SW1 ---
enable
configure terminal
hostname Core-SW1
no ip domain-lookup

vlan 10
 name Management
vlan 20
 name Servers
vlan 30
 name Office-LAN
vlan 40
 name WiFi
exit

! EtherChannel (Port-channel 1) Core-SW1 <-> Core-SW2 között
interface range FastEthernet0/1 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
 channel-group 1 mode active
 no shutdown
exit

! Trunk portok a routerek (R1-BP1, R1-BP2) és Access-SW-BP felé
interface range GigabitEthernet0/1 - 2, FastEthernet0/3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
 no shutdown
exit

! Szerverfarm access portok (VLAN 20 - WinSrv-DC és LinuxSrv)
interface range FastEthernet0/10 - 12
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
exit

! Menedzsment SVI és SSH alapok
interface vlan 10
 ip address 10.10.10.11 255.255.255.192
 no shutdown
exit
ip default-gateway 10.10.10.1

spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40 root primary
end
write memory

! --- CORE-SW2 ---
enable
configure terminal
hostname Core-SW2
no ip domain-lookup

vlan 10
 name Management
vlan 20
 name Servers
vlan 30
 name Office-LAN
vlan 40
 name WiFi
exit

interface range FastEthernet0/1 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
 channel-group 1 mode active
 no shutdown
exit

interface range GigabitEthernet0/1 - 2, FastEthernet0/3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
 no shutdown
exit

interface vlan 10
 ip address 10.10.10.12 255.255.255.192
 no shutdown
exit
ip default-gateway 10.10.10.1

spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40 root secondary
end
write memory
