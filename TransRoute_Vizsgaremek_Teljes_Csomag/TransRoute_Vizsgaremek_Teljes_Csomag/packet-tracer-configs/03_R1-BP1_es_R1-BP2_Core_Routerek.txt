! --- R1-BP1 (ACTIVE HSRP) ---
enable
configure terminal
hostname R1-BP1
ip domain-name transroute.local
crypto key generate rsa general-keys modulus 1024
username admin privilege 15 secret Cisco123!
ip ssh version 2

line vty 0 4
 login local
 transport input ssh
exit

ip routing
ipv6 unicast-routing

! LAN alinterfészek (Core-SW1 felé - Gi0/0/0)
interface GigabitEthernet0/0/0
 no shutdown
exit

interface GigabitEthernet0/0/0.10
 encapsulation dot1Q 10
 ip address 10.10.10.2 255.255.255.192
 ipv6 address 2001:DB8:100:10::2/64
 standby 10 ip 10.10.10.1
 standby 10 priority 110
 standby 10 preempt
 ipv6 ospf 1 area 0
exit

! ASA-BP tűzfal felé menő belső tranzit link (Gi0/0/1)
interface GigabitEthernet0/0/1

 ipv6 address 2001:DB8:100:254::1/64
 ipv6 ospf 1 area 0

exit

! OSPFv2 (IPv4)
router ospf 1
 router-id 1.1.1.1
 network 10.10.10.0 0.0.0.63 area 0
 network 10.10.20.0 0.0.0.63 area 0
 network 10.10.30.0 0.0.0.255 area 0
 network 10.10.40.0 0.0.0.255 area 0
 network 10.10.254.0 0.0.0.3 area 0
 default-information originate
exit

! OSPFv3 (IPv6)
ipv6 router ospf 1
 router-id 1.1.1.1
exit
end
write memory
